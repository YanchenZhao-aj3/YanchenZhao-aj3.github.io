### 第二步：编写核心调用代码

创建`index.js`文件，把以下代码复制进去。这段代码经过我亲测，100%跑通Grok4Fast模型。

javascript
import OpenAI from 'openai';

// 1. 实例化OpenAI客户端
// 这是最关键的一步：把baseURL改成[[云雾ai官网](https://www.yunwuai.cc/)](https://www.yunwuai.cc/)的接口地址
const client = new OpenAI({
  apiKey: '你的云雾API密钥', // 替换成你在云雾后台生成的Key
  baseURL: 'https://www.yunwuai.cc/v1', // 无需翻墙的国内直连地址
});

async function main() {
  try {
    // 2. 发起对话请求
    // 模型名称填入Grok4Fast对应的模型标识
    const response = await client.chat.completions.create({
      model: 'grok-4-fast', // 注意：模型名称请以云雾官网实际命名为准
      messages: [
        { role: 'system', content: '你是一个拥有顶级推理能力的AI助手。' },
        { role: 'user', content: '用中文解释一下量子纠缠的主要原理。' },
      ],
      temperature: 0.7,
      max_tokens: 1500, // 控制输出长度
      stream: false, // 我们先用非流式输出，方便新手理解
    });

    // 3. 输出结果
    console.log('=== 调用成功！Grok4Fast 回复如下 ===');
    console.log(response.choices[0].message.content);

  } catch (error) {
    // 4. 如果出错，打印并分析错误
    console.error('调用失败，错误信息：', error.message);
    if (error.status) {
      console.error('HTTP状态码：', error.status);
    }
  }
}

main();

## 跑得更高级：流式输出

上面的示例是非流式的，你会等很久才一次性看到全部回复。如果想让体验更好，使用流式输出，像ChatGPT那样一个字一个字地出现，你可以把`stream`设为`true`，然后稍微改一下代码：

javascript
const stream = await client.chat.completions.create({
  model: 'grok-4-fast',
  messages: [
    { role: 'user', content: '写一首关于代码的现代诗，要短。' },
  ],
  stream: true, // 开启流式输出
});

for await (const chunk of stream) {
  process.stdout.write(chunk.choices[0]?.delta?.content || '');
}


运行这段代码，你会发现回复像打字机一样在终端里“打”出来，体验感直接拉满。
