const axios = require('axios');
const { createHash } = require('crypto');

async function chatWithDeepSeek(apiKey, prompt) {
    // 1. 生成签名
    const timestamp = Date.now().toString();
    const sign = createHash('sha256').update(apiKey + timestamp).digest('hex');

    // 2. 构造请求头
    const headers = {
        'Authorization': `Bearer ${apiKey}`,
        'X-Request-Timestamp': timestamp,
        'X-Request-Sign': sign,
        'Content-Type': 'application/json'
    };

    // 3. 构造请求体，注意模型名、参数格式
    const body = {
        model: 'deepseek-chat',
        messages: [{ role: 'user', content: prompt }],
        stream: true,
        max_tokens: 2048
    };

    // 4. 发送请求并处理流式响应
    const response = await axios.post('https://api.deepseek.com/v1/chat/completions', body, { headers, responseType: 'stream' });
    
    // ...（后续复杂的流式数据解析、错误处理、连接释放等，往往需要额外100+行）
    
    return response.data;
