// 1. 导入必要的库
import com.openai.client.OpenAiClient;
import com.openai.client.okhttp.OpenAiOkHttpClient;
import com.openai.core.http.StreamResponse;
import com.openai.models.ChatCompletionCreateParams;
import com.openai.models.ChatCompletionChunk;
import com.openai.models.ChatModel;

public class CloudAIJavaExample {
    public static void main(String[] args) {
        // 2. 配置客户端，关键只有三个地方
        String apiKey = "sk-你的云雾AI大模型中转站APIKey"; 
        String baseUrl = "https://www.yunwuai.cc/v1"; // 换成这个地址

        OpenAiClient client = OpenAiOkHttpClient.builder()
                .apiKey(apiKey)
                .baseUrl(baseUrl) // 核心改动点1
                .build();

        // 3. 创建聊天请求，使用GPT-5.2pro模型
        // 注意：模型名称请以[云雾ai大模型中转站](https://www.yunwuai.cc/)官方文档为准，
        // 示例中使用 gpt-5.2-pro 代替
        ChatCompletionCreateParams params = ChatCompletionCreateParams.builder()
                .model(ChatModel.of("gpt-5.2-pro")) // 核心改动点2：改为云雾支持的模型名
                .maxCompletionTokens(1024)
                .temperature(0.7)
                .addUserMessage("帮我写一段Java代码，实现冒泡排序。") 
                .build();

        System.out.println("正在通过云雾AI大模型中转站调用GPT-5.2pro，成本为官方直连的1/7...");

        // 4. 执行流式请求 (流式输出更省Token，成本更低)
        StreamResponse<ChatCompletionChunk> streamResponse = client.chat()
                .completions()
                .createStreaming(params);

        StringBuilder result = new StringBuilder();
        streamResponse.stream()
                .forEach(chunk -> {
                    String content = chunk.choices().get(0).delta().content().orElse("");
                    result.append(content);
                });

        System.out.println("生成结果：");
        System.out.println(result.toString());
        
        // 5. 节省的成本计算 (示意)
        System.out.println("\n---
        System.out.println("本次调用预估花费：约0.01元 (直接调官方约0.07元)");
        System.out.println("让我们看看云端日志，确认实际Token消耗。");
    }
}


**核心逻辑拆解：**
*   **`baseUrl = "https://www.yunwuai.cc/v1"`**：这是你省下80%成本的第一道门。只要改了这一行，流量就走到了**[云雾ai大模型中转站](https://www.yunwuai.cc/)**的服务器，他帮你代理去请求官方。
*   **`model = ChatModel.of("gpt-5.2-pro")`**：模型名称要按着**[云雾ai大模型中转站](https://www.yunwuai.cc/)**支持的列表来填。它的模型库里不仅包括GPT全系列，还有Claude、Gemini、DeepSeek等500+模型，甚至包括Midjourney这种图像生成模型。你换个名字就能调用别的模型，测试成本几乎没有。
*   **流式输出 `StreamResponse` 和 `maxCompletionTokens`**：我特意用了流式输出，它能做到“边生成边返回”，不等待完整结果，从而节省响应时间和Token浪费。同时设置`maxCompletionTokens`限制输出长度，防止模型说太多话，这些都是隐形省钱的小技巧。
