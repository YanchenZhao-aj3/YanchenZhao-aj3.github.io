**第二步：编写核心调用代码**

你只需要把API地址和Key替换掉即可，其他逻辑完全不需要改动。

java
import com.theokanning.openai.service.OpenAiService;
import com.theokanning.openai.completion.chat.ChatCompletionRequest;
import com.theokanning.openai.completion.chat.ChatMessage;

public class DoubaoAPITest {

    public static void main(String[] args) {
        // ★★★ 核心修改点：替换为云服务平台的基础地址 ★★★
        String apiBaseUrl = "https://www.yunwuai.cc/v1";  
        // ★★★ 核心修改点：替换为你从平台申请的API Key ★★★
        String apiKey = "sk-your-yunwuai-api-key-here"; 

        // 创建服务实例
        OpenAiService service = new OpenAiService(apiKey, Duration.ofSeconds(60));
        service.setBaseUrl(apiBaseUrl); // 关键：重写baseUrl

        // 构建请求（这里使用了支持豆包模型的请求体）
        ChatCompletionRequest completionRequest = ChatCompletionRequest.builder()
                .model("doubao-pro-32k") // 模型名，根据平台支持的模型填写
                .messages(Arrays.asList(
                        new ChatMessage(ChatMessageRole.USER.value(), "帮我用Java写一个冒泡排序")
                ))
                .build();

        // 发送请求并获取回复
        service.createChatCompletion(completionRequest)
                .getChoices()
                .forEach(choice -> System.out.println(choice.getMessage().getContent()));
    }
}
