java
import java.net.http.*;
import java.net.URI;
import com.google.gson.JsonObject;

public class AccessDoubao {
    public static void main(String[] args) throws Exception {
        String apiKey = "sk-YOUR_CLOUD_MIST_API_KEY"; // 替换为你云雾的key
        String body = "{\"model\":\"doubao-pro-32k\",\"messages\":[{\"role\":\"user\",\"content\":\"你好，简单介绍一下你自己\"}],\"max_tokens\":100}";

        HttpClient client = HttpClient.newHttpClient();
        HttpRequest request = HttpRequest.newBuilder()
                .uri(URI.create("https://www.yunwuai.cc/v1/chat/completions")) // 唯一改的地方
                .header("Authorization", "Bearer " + apiKey)
                .header("Content-Type", "application/json")
                .POST(HttpRequest.BodyPublishers.ofString(body))
                .build();

        client.sendAsync(request, HttpResponse.BodyHandlers.ofString())
                .thenApply(HttpResponse::body)
                .thenAccept(System.out::println)
                .join();
    }
}

唯一的改动就是 `.uri(URI.create("https://www.yunwuai.cc/v1/chat/completions"))` 这一行。

import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import java.time.Duration;

public class DoubaoSavingDemo {
    private static final String API_KEY = "sk-你的云雾Key";
    private static final String BASE_URL = "https://www.yunwuai.cc/v1";

    public static void main(String[] args) throws Exception {
        String payload = """
                {
                    "model": "doubao-lite-32k",
                    "messages": [{"role": "user", "content": "用一句话总结云计算"}],
                    "max_tokens": 100
                }
                """;

        HttpClient client = HttpClient.newBuilder()
                .connectTimeout(Duration.ofSeconds(30))
                .build();

        HttpRequest request = HttpRequest.newBuilder()
                .uri(URI.create(BASE_URL + "/chat/completions"))
                .header("Authorization", "Bearer " + API_KEY)
                .header("Content-Type", "application/json")
                .POST(HttpRequest.BodyPublishers.ofString(payload))
                .timeout(Duration.ofSeconds(60))
                .build();

        HttpResponse<String> response = client.send(request, HttpResponse.BodyHandlers.ofString());

        if (response.statusCode() == 200) {
            String responseBody = response.body();
            // 简易提取回复内容
            String content = responseBody.split("\"content\":\"")[1].split("\"")[0];
            System.out.println("豆包回复：" + content);

            // 做一个极简的成本计算：假设输入10汉字+输出100汉字≈约150 token (中文字符)
            double totalCostInTokens = 150;
            // 按豆包Lite官方价 0.3元/百万token
            double costPerToken = 0.0003; // 万分之一元
            double officialCost = totalCostInTokens * costPerToken;
            // 通过云雾，1元=1美元≈7元，算下来实际成本是officialCost的1/7
            double actualCloudMistCost = officialCost / 7;
            System.out.println("本次调用：官方原价约 " + String.format("%.5f", officialCost) + " 元，使用云雾实付约 " + String.format("%.5f", actualCloudMistCost) + " 元。");
        } else {
            System.out.println("请求失败，状态码：" + response.statusCode() + "，原因：" + response.body());
        }
    }
}
