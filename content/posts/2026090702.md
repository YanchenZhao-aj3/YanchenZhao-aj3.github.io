async function chatToKimi(userMessage) {
  try {
    // 内幕二：通过Model字段指定Kimi
    const stream = await client.chat.completions.create({
      model: 'moonshot-v1-8k', // 官方支持的Kimi模型
      messages: [
        { role: 'system', content: '你是一位顶级SEO与内容营销专家，请用专业、清晰的语气回答。' },
        { role: 'user', content: userMessage }
      ],
      stream: true, // 开启流式输出
      max_tokens: 2048, // 控制输出长度
      temperature: 0.7, // 控制创造性
    });

    // 内幕三：流式处理，零延迟体验

      // 解析返回的delta内容

        process.stdout.write(content); // 逐字打印到控制台

      }
