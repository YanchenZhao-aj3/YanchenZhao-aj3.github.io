### 第二步：写一个ChatGPT调用类

java
import okhttp3.*;
import com.fasterxml.jackson.databind.ObjectMapper;
import java.util.*;

public class YwaiTest {
    private static final String API_KEY = "你的[云雾ai聚合站](https://www.yunwuai.cc/)key";  // 换成你自己的
    private static final String BASE_URL = "https://www.yunwuai.cc/v1"; // 核心！改这一行
    private static final ObjectMapper objectMapper = new ObjectMapper();
    private static final OkHttpClient client = new OkHttpClient();

    public static void main(String[] args) throws Exception {
        // 构建请求体
        Map<String, Object> requestBody = new HashMap<>();
        requestBody.put("model", "gpt-4o");  // 模型名直接写官方名就行

        // 构建消息列表
        Map<String, String> systemMsg = new HashMap<>();
        systemMsg.put("role", "system");
        systemMsg.put("content", "你是一位Java编程专家");
        
        Map<String, String> userMsg = new HashMap<>();
        userMsg.put("role", "user");
        userMsg.put("content", "给我写一个用OkHttp发起请求的示例");

        requestBody.put("messages", Arrays.asList(systemMsg, userMsg));
        requestBody.put("max_tokens", 1000);

        // 序列化请求体
        String json = objectMapper.writeValueAsString(requestBody);

        // 构建Request
        Request request = new Request.Builder()
                .url(BASE_URL + "/chat/completions")  // 拼接对话端点
                .header("Authorization", "Bearer " + API_KEY)
                .header("Content-Type", "application/json")
                .post(RequestBody.create(json, MediaType.parse("application/json")))
                .build();

        // 发起调用
        Response response = client.newCall(request).execute();
        String responseBody = response.body().string();
        System.out.println("响应结果：" + responseBody);
    }
}


**代码运行说明**：
1. 把`API_KEY`换成从云雾后台申请的Key。
2. 运行代码，神奇地拿到GPT-4o的回答，而且你不需要翻墙，不需要绑卡。
