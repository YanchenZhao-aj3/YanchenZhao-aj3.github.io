### 第三步：主代码文件 `index.js`

javascript
import OpenAI from 'openai';
import dotenv from 'dotenv';
dotenv.config();

const client = new OpenAI({
  baseURL: 'https://www.yunwuai.cc/v1',
  apiKey: process.env.YUNWU_API_KEY,
});

async function askO4Mini(question) {
  try {
    const completion = await client.chat.completions.create({
      model: 'o4-mini',   // 关键：直接指定模型名
      messages: [
        { role: 'system', content: '你是一个友好的AI助手，回答要简洁准确。' },
        { role: 'user', content: question },
      ],
      max_tokens: 1024,    // 控制成本，o4-mini省Token
      temperature: 0.7,
    });

    return completion.choices[0].message.content;
  } catch (error) {
    console.error('调用失败：', error);
    return null;
  }
}

// 测试调用
(async () => {
  const answer = await askO4Mini('帮我用三句话解释“量子纠缠”');
  console.log('AI回答：', answer);
})();
