    // 云雾AI平台的核心接入点（完全兼容OpenAI标准）

    private static final String API_KEY = "你的云雾API_KEY_这里替换";

    private static final OkHttpClient client = new OkHttpClient();
    private static final ObjectMapper mapper = new ObjectMapper();

    // 核心方法：统一发送消息给指定模型
    public static String sendMessage(String model, String userMessage) throws IOException {
        // 1. 构建符合OpenAI规范的请求体
        String jsonBody = "{"
                + "\"model\": \"" + model + "\","
                + "\"messages\": ["
                + "{\"role\": \"user\", \"content\": \"" + userMessage.replaceAll("\"", "\\\\\"") + "\"}"
                + "]"
                + "}";

        // 2. 构造请求
        RequestBody body = RequestBody.create(jsonBody, MediaType.parse("application/json; charset=utf-8"));
        Request request = new Request.Builder()
                .url(BASE_URL + "/chat/completions")  // 直接调用云雾的/v1端点
                .post(body)
                .addHeader("Authorization", "Bearer " + API_KEY)
                .addHeader("Content-Type", "application/json")
                .build();

        // 3. 发送并解析响应
        try (Response response = client.newCall(request).execute()) {
            if (!response.isSuccessful()) {
                System.err.println("请求失败: " + response.code());
                return null;
            }
            String rawResponse = response.body().string();
            // 简单解析出content（生产环境建议用完整JsonNode解析）
            return mapper.readTree(rawResponse)
                    .path("choices")
                    .get(0)
                    .path("message")
                    .path("content")
                    .asText();
        }
    }

    public static void main(String[] args) throws IOException {
        // 示例1：调用GPT-4o进行分析

        System.out.println("GPT-4o回复: " + reply1);
        
        // 示例2：调用Claude 3.5 Sonnet进行创意写作

        System.out.println("Claude 3.5 Sonnet回复: " + reply2);
