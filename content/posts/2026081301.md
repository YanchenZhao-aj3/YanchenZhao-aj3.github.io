async function callDoubaoStream() {
  const stream = await client.chat.completions.create({
    model: 'doubao-pro-32k',
    messages: [
      { role: 'user', content: '请写一首赞美Node.js异步IO的诗。' }
    ],
    stream: true,
  });

  let result = '';
  for await (const chunk of stream) {

}
