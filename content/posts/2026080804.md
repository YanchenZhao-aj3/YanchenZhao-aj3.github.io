### 第二步：写一个极其简单的Java调用代码

我们使用Java中最常用的HTTP客户端 `OkHttp` 和JSON库 `Gson`。如果你不想引入依赖，也可以用原生的 `HttpURLConnection`，这里用OkHttp是为了代码更简洁。

**完整代码示例（复制就能用）：**

java
import okhttp3.*;
import com.google.gson.JsonObject;
import com.google.gson.JsonParser;
import java.io.IOException;

public class ClaudeJavaDemo {
    public static void main(String[] args) throws IOException {
        // 1. 把你的云雾API Key替换到这里
        String apiKey = "sk-你的云雾API Key";
        
        // 2. 调用地址：必须用云雾的API接口
        String url = "https://www.yunwuai.cc/v1/chat/completions";

        // 3. 构建请求体（OpenAI兼容格式）
        JsonObject message = new JsonObject();
        message.addProperty("role", "user");
        message.addProperty("content", "用中文介绍一下你自己，并说一句话证明你在思考。");

        JsonObject body = new JsonObject();
        body.addProperty("model", "claude-3-5-sonnet-20241022"); // 这里指定Claude模型
        body.add("messages", new com.google.gson.JsonArray());
        body.getAsJsonArray("messages").add(message);

        // 4. 创建OkHttp请求
        OkHttpClient client = new OkHttpClient();
        Request request = new Request.Builder()
                .url(url)
                .header("Authorization", "Bearer " + apiKey)
                .header("Content-Type", "application/json")
                .post(RequestBody.create(body.toString(), MediaType.parse("application/json")))
                .build();

        // 5. 发送请求并获取响应
        Response response = client.newCall(request).execute();
        String responseBody = response.body().string();

        // 6. 解析并输出结果
        JsonObject jsonResponse = JsonParser.parseString(responseBody).getAsJsonObject();
        String assistantReply = jsonResponse.getAsJsonArray("choices")
                .get(0).getAsJsonObject()
                .getAsJsonObject("message")
                .get("content")
                .getAsString();

        System.out.println("Claude回复：");
        System.out.println(assistantReply);
    }
}
