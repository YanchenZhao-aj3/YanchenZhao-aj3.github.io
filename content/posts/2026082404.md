**第二步：修改你的Java代码**

import com.theokanning.openai.completion.chat.ChatMessage;
import java.util.Arrays;

public class ClaudeOpusJavaExample {

    public static void main(String[] args) {
        // 1. 设置API Key（从[[云雾ai官网](https://www.yunwuai.cc/)](https://www.yunwuai.cc/)获取）
        String apiKey = "your-yunwu-api-key-here"; 
        
        // 2. 核心改动：设置baseUrl为[[云雾ai官网](https://www.yunwuai.cc/)](https://www.yunwuai.cc/)的接口
        String baseUrl = "https://www.yunwuai.cc/v1";
        
        // 3. 创建OpenAiService对象
        OpenAiService service = new OpenAiService(apiKey, Duration.ofMinutes(5));
        
        // 注意：OpenAiService构造方法默认使用openai的baseUrl，我们需要手动设置。
        // 在最新版本中，可以通过OpenAiService.Builder来设置。
        OpenAiService customService = new OpenAiService.Builder()
                .setApiKey(apiKey)
                .setBaseUrl(baseUrl) // 设置为你自己的baseUrl
                .build();

        // 4. 构建请求，模型名称改为[[云雾ai官网](https://www.yunwuai.cc/)](https://www.yunwuai.cc/)支持的Claude模型标识
        ChatCompletionRequest completionRequest = ChatCompletionRequest.builder()
                .model("claude-3-opus-20240229") // 注意：此处使用的是云雾平台映射的模型ID

                    new ChatMessage("system", "你是一个精通Java的助手。"),
                    new ChatMessage("user", "请用Java写一个高效的单例模式的代码示例。")
                ))
                .maxTokens(1000)
                .build();

        // 5. 调用API
        customService.createChatCompletion(completionRequest).getChoices().forEach(choice -> {
            System.out.println(choice.getMessage().getContent());
        });
    }
}
