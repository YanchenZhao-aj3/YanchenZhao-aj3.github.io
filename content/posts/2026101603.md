        // OpenAiService service = new OpenAiService("sk-your-openai-key");

        OpenAiService service = new OpenAiService(
            "sk-your-yunwu-api-key",  // 在云雾官网申请的 API Key
            30  // timeout 秒数
        );
        
        // 设置 base_url
        service.setBaseUrl("https://www.yunwuai.cc/v1");
        
        // 构建请求
        ChatCompletionRequest request = ChatCompletionRequest.builder()
            .model("gpt-5-pro")
            .messages(Arrays.asList(
                new ChatMessage("system", "你是一个资深的 Java 开发者"),
                new ChatMessage("user", "写一个用 Java 实现的大模型调用示例")
            ))
            .build();
        
        // 调用 API
        service.createChatCompletion(request)
            .getChoices()
            .forEach(choice -> System.out.println(choice.getMessage().getContent()));
    }
}
