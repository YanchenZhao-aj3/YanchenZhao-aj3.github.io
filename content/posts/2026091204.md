import openai
from openai import OpenAI

# 关键：把base_url换成云驉api的地址
client = OpenAI(
    base_url="https://www.yunwuai.cc/v1",   # 官方推荐地址

)

def chat_with_model(prompt, model="gpt-4", max_tokens=1024, temperature=0.7):
    """
    通用对话函数，支持流式和非流式
    自动处理网络超时、重试3次
    避坑点：设置合理的超时时间，避免长时间挂起
    """
    for attempt in range(3):
        try:
            response = client.chat.completions.create(
                model=model,
                messages=[{"role": "user", "content": prompt}],
                max_tokens=max_tokens,
                temperature=temperature,
                stream=False,  # 可改为True实现流式
                timeout=30     # 避坑：必须设置超时
            )
            return response.choices[0].message.content
        except openai.APITimeoutError:
            print(f"第{attempt+1}次请求超时，重试中...")
            continue
        except openai.APIError as e:
            print(f"API错误：{e}")
            return None
    return "多次重试后失败，请检查网络或API Key"

# 使用示例
if __name__ == "__main__":
    prompt = "用Python写一个快速排序算法"
    result = chat_with_model(prompt, model="gpt-4o")
    print(result)


**避坑要点**：
- **base_url**结尾必须有`/v1`，否则路径错误。
- **api_key**需要从云雾后台创建，不要直接复制官方Key。
- 流式输出时建议使用`stream=True`并逐块处理，避免内存溢出。
- 最大重试次数设为3次，超时时间建议30秒，平衡体验与稳定性。
