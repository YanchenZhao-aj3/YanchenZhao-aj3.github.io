### 第二步：编写Java调用代码
下面是一段完整的Java代码。你只需要替换掉 `API_KEY_HERE` 这个占位符，其他部分直接复制就能用。

java
import okhttp3.*;
import com.google.gson.Gson;
import com.google.gson.JsonArray;
import com.google.gson.JsonObject;
import java.io.IOException;

public class Gpt5CodexExample {
    // 替换为你在[云雾ai聚合站](https://www.yunwuai.cc/)申请的 API Key
    private static final String API_KEY = "API_KEY_HERE";
    
    // 使用[云雾ai聚合站](https://www.yunwuai.cc/)提供的 API 地址
    private static final String API_URL = "https://www.yunwuai.cc/v1/chat/completions";

    public static void main(String[] args) throws IOException {
        // 1. 构造请求体
        JsonObject requestBody = new JsonObject();
        requestBody.addProperty("model", "gpt-5-codex"); // 指定模型
        requestBody.addProperty("max_tokens", 1024);     // 最大输出长度
        requestBody.addProperty("temperature", 0.7);     // 创造性
        
        // 构造消息列表
        JsonArray messages = new JsonArray();
        
        // 系统角色设定
        JsonObject systemMessage = new JsonObject();
        systemMessage.addProperty("role", "system");
        systemMessage.addProperty("content", "你是一个资深的Java代码专家。");
        messages.add(systemMessage);
        
        // 用户提问
        JsonObject userMessage = new JsonObject();
        userMessage.addProperty("role", "user");
        userMessage.addProperty("content", "写一个Spring Boot的REST Controller，用于实现用户登录接口，使用JWT验证Token。");
        messages.add(userMessage);
        
        requestBody.add("messages", messages);

        // 2. 构建HTTP请求
        OkHttpClient client = new OkHttpClient();
        Request request = new Request.Builder()
                .url(API_URL)
                .addHeader("Content-Type", "application/json")
                .addHeader("Authorization", "Bearer " + API_KEY)
                .post(RequestBody.create(
                        MediaType.parse("application/json; charset=utf-8"),
                        new Gson().toJson(requestBody)
                ))
                .build();

        // 3. 发送请求并获取响应
        try (Response response = client.newCall(request).execute()) {
            if (response.isSuccessful() && response.body() != null) {
                String responseString = response.body().string();
                System.out.println("GPT-5 Codex返回的代码：");
                System.out.println(responseString);
            } else {
                System.out.println("请求失败，状态码：" + response.code());
                System.out.println("响应体内容：" + (response.body() != null ? response.body().string() : "空"));
            }
        }
    }
}
