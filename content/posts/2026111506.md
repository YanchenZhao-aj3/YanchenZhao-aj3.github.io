    response = client.chat.completions.create(

        messages=[

                "role": "user",
                "content": "请用一段简洁的中文介绍什么是KimiK2大模型，并说明用它写代码的优势。"
            }
        ],
        # 其他参数可以按需设置，比如温度、最大Token数等
